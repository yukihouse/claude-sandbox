1"""Helpers that hand-assemble minimal, synthetic PE (.exe) byte strings for 

2tests, so the parser can be exercised without needing a real Windows 

3toolchain to compile fixtures. 

4"""

6from __future__ import annotations

8import struct

10IMAGE_FILE_MACHINE_I386 = 0x14C 

11OPTIONAL_HEADER_SIZE = 224 # 96 fixed bytes + 16 data-directory entries * 8 bytes 

12DOS_HEADER_SIZE = 64 

13SECTION_HEADER_SIZE = 40 

14 

15# Fixed header size when the file has exactly one section: DOS header + 

16# "PE\0\0" + COFF file header + optional header + one section header. 

17HEADER_SIZE = DOS_HEADER_SIZE + 4 + 20 + OPTIONAL_HEADER_SIZE + SECTION_HEADER_SIZE 

18 

19VS_VERSION_INFO_KEY = "VS_VERSION_INFO\x00".encode("utf-16-le")

22def _dos_header() -> bytes: 

23 header = bytearray(DOS_HEADER_SIZE) 

24 header[0:2] = b"MZ" 

25 struct.pack_into("<I", header, 0x3C, DOS_HEADER_SIZE) 

26 return bytes(header)

29def _optional_header32(data_directory: list[tuple[int, int]]) -> bytes: 

30 fixed = struct.pack( 

31 "<HBBIIIIIIIIIHHHHHHIIIIHHIIIIII", 

32 0x10B, # Magic (PE32) 

33 0, 

34 0, # Major/MinorLinkerVersion 

35 0, # SizeOfCode 

36 0, # SizeOfInitializedData 

37 0, # SizeOfUninitializedData 

38 0, # AddressOfEntryPoint 

39 0, # BaseOfCode 

40 0, # BaseOfData (PE32 only) 

41 0x400000, # ImageBase 

42 0x1000, # SectionAlignment 

43 0x200, # FileAlignment 

44 0, 

45 0, # Major/MinorOperatingSystemVersion 

46 0, 

47 0, # Major/MinorImageVersion 

48 0, 

49 0, # Major/MinorSubsystemVersion 

50 0, # Win32VersionValue 

51 0, # SizeOfImage 

52 0, # SizeOfHeaders 

53 0, # CheckSum 

54 2, # Subsystem 

55 0, # DllCharacteristics 

56 0, 

57 0, 

58 0, 

59 0, # Stack/Heap reserve/commit 

60 0, # LoaderFlags 

61 len(data_directory), # NumberOfRvaAndSizes 

62 ) 

63 directory = b"".join(struct.pack("<II", va, size) for va, size in data_directory) 

64 header = fixed + directory 

65 assert len(header) == OPTIONAL_HEADER_SIZE 

66 return header

69def _section_header( 

70 name: bytes, 

71 virtual_size: int, 

72 virtual_address: int, 

73 size_of_raw_data: int, 

74 pointer_to_raw_data: int, 

75) -> bytes: 

76 header = name.ljust(8, b"\x00")[:8] + struct.pack( 

77 "<IIIIIIHHI", 

78 virtual_size, 

79 virtual_address, 

80 size_of_raw_data, 

81 pointer_to_raw_data, 

82 0, # PointerToRelocations 

83 0, # PointerToLinenumbers 

84 0, # NumberOfRelocations

86 0x40000040, # Characteristics (initialized data, readable) 

87 ) 

88 assert len(header) == SECTION_HEADER_SIZE 

89 return header

92def build_pe(resource_section: bytes | None) -> bytes: 

93 """Build a minimal, well-formed PE32 file. If `resource_section` is 

94 given, it becomes the raw bytes of a single ".rsrc" section, identity 

95 mapped (RVA == file offset) for simplicity, and wired up via the 

96 resource data directory entry.""" 

97 data_directory = [(0, 0)] * 16 

98 sections = b"" 

99 number_of_sections = 0 

100 

101 if resource_section is not None: 

102 number_of_sections = 1 

103 data_directory[2] = (HEADER_SIZE, len(resource_section)) 

104 sections = _section_header( 

105 b".rsrc", 

106 virtual_size=len(resource_section), 

107 virtual_address=HEADER_SIZE, 

108 size_of_raw_data=len(resource_section), 

109 pointer_to_raw_data=HEADER_SIZE, 

110 ) 

111 

112 file_header = struct.pack( 

113 "<HHIIIHH", 

114 IMAGE_FILE_MACHINE_I386, 

115 number_of_sections, 

116 0, # TimeDateStamp 

117 0, # PointerToSymbolTable 

118 0, # NumberOfSymbols 

119 OPTIONAL_HEADER_SIZE, 

120 0x0102, # Characteristics: EXECUTABLE_IMAGE | 32BIT_MACHINE 

121 ) 

122 

123 pe = _dos_header() + b"PE\x00\x00" + file_header + _optional_header32(data_directory) + sections 

124 if resource_section is not None: 

125 pe += resource_section 

126 assert len(pe) - len(resource_section) == HEADER_SIZE 

127 return pe

130def _resource_directory(entries: list[tuple[int, int]]) -> bytes: 

131 header = struct.pack("<IIHHHH", 0, 0, 0, 0, 0, len(entries)) 

132 body = b"".join( 

133 struct.pack("<II", name_or_id, offset_to_data) for name_or_id, offset_to_data in entries 

134 ) 

135 return header + body

138def build_resource_section_skips_non_version_leaf_then_finds_version_info() -> bytes: 

139 """Root -> name -> a language directory with two leaf entries: the 

140 first does not resolve to version info (zero-length data), the second 

141 does. Exercises the "keep scanning siblings" path.""" 

142 name_offset, lang_offset = 24, 48 

143 non_version_entry_offset, version_entry_offset, version_offset = 80, 96, 112 

144 

145 root_dir = _resource_directory([(16, name_offset | 0x80000000)]) 

146 name_dir = _resource_directory([(1, lang_offset | 0x80000000)]) 

147 lang_dir = _resource_directory( 

148 [(0x409, non_version_entry_offset), (0x40A, version_entry_offset)] 

149 ) 

150 

151 non_version_data_entry = struct.pack("<IIII", 0, 0, 0, 0) # version_size == 0 

152 

153 fixed_file_info = struct.pack("<I", 0xFEEF04BD) + b"\x00" * 48 

154 body = VS_VERSION_INFO_KEY + b"\x00\x00" + fixed_file_info 

155 version_data = struct.pack("<HHH", 6 + len(body), len(fixed_file_info), 0) + body 

156 resource_rva = HEADER_SIZE 

157 version_data_entry = struct.pack( 

158 "<IIII", resource_rva + version_offset, len(version_data), 0, 0 

159 ) 

160 

161 section = ( 

162 root_dir + name_dir + lang_dir + non_version_data_entry + version_data_entry + version_data 

163 ) 

164 assert len(section) == version_offset + len(version_data) 

165 return section

168def build_version_resource_section(resource_type_id: int = 16) -> bytes: 

169 """Build a ".rsrc" section containing a 3-level resource directory 

170 (type -> name -> language) with a single leaf resource of the given 

171 type ID. When `resource_type_id` is RT_VERSION (16, the default) the 

172 leaf's data is a minimal but valid VS_VERSIONINFO blob; otherwise it 

173 is unused filler, letting tests build a resource section that has no 

174 RT_VERSION entry at all. 

175 """ 

176 name_offset, lang_offset, data_entry_offset, version_offset = 24, 48, 72, 88 

177 

178 root_dir = _resource_directory([(resource_type_id, name_offset | 0x80000000)]) 

179 name_dir = _resource_directory([(1, lang_offset | 0x80000000)]) 

180 lang_dir = _resource_directory([(0x409, data_entry_offset)]) 

181 

182 fixed_file_info = struct.pack("<I", 0xFEEF04BD) + b"\x00" * 48 

183 assert len(fixed_file_info) == 52 

184 body = VS_VERSION_INFO_KEY + b"\x00\x00" + fixed_file_info # pad key to 4-byte boundary 

185 version_data = struct.pack("<HHH", 6 + len(body), len(fixed_file_info), 0) + body 

186 

187 resource_rva = HEADER_SIZE 

188 data_entry = struct.pack("<IIII", resource_rva + version_offset, len(version_data), 0, 0) 

189 

190 section = root_dir + name_dir + lang_dir + data_entry + version_data 

191 assert len(section) == version_offset + len(version_data) 

192 return section
