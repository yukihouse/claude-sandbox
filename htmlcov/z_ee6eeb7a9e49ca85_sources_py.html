109def reader_temperature(station: str) -> Dataset: 

110 """Monthly mean surface air temperature for a SCAR READER station.""" 

111 if station not in READER_STATIONS: 

112 raise KeyError(f"unknown READER station: {station}") 

113 return Dataset( 

114 key=f"reader_{station}", 

115 title=f"{READER_STATIONS[station]} 月平均気温", 

116 provider="SCAR READER / British Antarctic Survey", 

117 url=READER_URL_TEMPLATE.format(station=station), 

118 homepage="https://legacy.bas.ac.uk/met/READER/", 

119 description="有人観測基地の地上気象観測から作成された月平均気温 (°C)。", 

120 citation=("Turner, J. et al. (2004) The SCAR READER project, J. Climate 17, 2890-2898."), 

121 )

133class FetchError(RuntimeError): 

134 """Raised when a dataset can be neither downloaded nor read from cache.""" 

135 

136 

137Opener = Callable[..., object] 

138 

139 

140def cache_path(url: str, cache_dir: Path) -> Path: 

141 digest = hashlib.sha256(url.encode("utf-8")).hexdigest()[:16] 

142 return cache_dir / f"{digest}.txt" 

143 

144 

145def fetch_text( 

146 url: str, 

147 *, 

148 cache_dir: Path = DEFAULT_CACHE_DIR, 

149 max_age: float = DEFAULT_MAX_AGE, 

150 opener: Opener = urllib.request.urlopen, 

151 now: Callable[[], float] = time.time, 

152 timeout: float = 30, 

153) -> str: 

154 """Return the body of ``url``, using an on-disk cache younger than ``max_age``. 

155 

156 If the download fails, a stale cache entry is returned instead so the UI keeps 

157 working offline; only when there is no cache at all is ``FetchError`` raised. 

158 """ 

159 path = cache_path(url, cache_dir) 

160 if path.exists() and now() - path.stat().st_mtime < max_age: 

161 return path.read_text(encoding="utf-8") 

162 

163 request = urllib.request.Request(url, headers={"User-Agent": USER_AGENT}) 

164 try: 

165 with opener(request, timeout=timeout) as response: 

166 text = response.read().decode("utf-8", errors="replace") 

167 except OSError as exc: # URLError and timeouts are OSErrors 

168 if path.exists(): 

169 return path.read_text(encoding="utf-8") 

170 raise FetchError(f"{url} を取得できませんでした: {exc}") from exc 

171 

172 path.parent.mkdir(parents=True, exist_ok=True) 

173 path.write_text(text, encoding="utf-8") 

174 return text
